```typescript
import { z } from 'zod';
import { shellRequestSchema } from './shell.js';

/**
 * Broad intent categories classified before any context gathering.
 * Kept intentionally coarse for MVP — see RESEARCH.md for rationale.
 */
export const requestIntentSchema = z.enum([
  'shell-command',
  'codebase',
  'filesystem',
  'general',
  'unknown',
]);

export type RequestIntent = z.infer<typeof requestIntentSchema>;

/**
 * Normalized internal request — shell transport plus inferred intent.
 * Consumers downstream of the shell bridge work with this type, not ShellRequest.
 */
export const normalizedRequestSchema = shellRequestSchema.extend({
  intent: requestIntentSchema,
});

export type NormalizedRequest = z.infer<typeof normalizedRequestSchema>;

/**
 * Always-present base context derived from the normalized request and
 * process globals. Never requires I/O.
 */
export const baseContextSchema = z.object({
  queryText: z.string(),
  cwd: z.string(),
  ttyPath: z.string(),
  shellPid: z.number().int().positive(),
  /** Hardcoded 'zsh' for Phase 2 — cross-shell support is v2 scope. */
  shellName: z.literal('zsh'),
  platform: z.string(),
  timestamp: z.string(),
});

export type BaseContext = z.infer<typeof baseContextSchema>;

/**
 * Result of the intent classifier. Includes signals for debugging
 * why a request was routed to a particular intent.
 */
export const intentDecisionSchema = z.object({
  intent: requestIntentSchema,
  confidence: z.number().min(0).max(1),
  signals: z.array(z.string()),
});

export type IntentDecision = z.infer<typeof intentDecisionSchema>;

/**
 * A typed chunk of extra context gathered by an intent-gated provider.
 * Discriminated on `kind` so consumers can narrow safely.
 */
export const contextChunkSchema = z.discriminatedUnion('kind', [
  z.object({
    kind: z.literal('git'),
    payload: z.object({
      cwd: z.string(),
      root: z.string(),
      branch: z.string().nullable(),
      dirty: z.boolean(),
      changedFiles: z.array(z.string()),
    }),
  }),
  z.object({
    kind: z.literal('filesystem'),
    payload: z.object({
      cwd: z.string(),
      apparentFilename: z.string().nullable(),
    }),
  }),
]);

export type ContextChunk = z.infer<typeof contextChunkSchema>;

/**
 * Fully assembled context passed to the provider adapter.
 * base is always populated; extras contains only intent-relevant chunks.
 */
export const contextEnvelopeSchema = z.object({
  base: baseContextSchema,
  extras: z.array(contextChunkSchema),
});

export type ContextEnvelope = z.infer<typeof contextEnvelopeSchema>;
```

```typescript
import { type NormalizedRequest, type IntentDecision, requestIntentSchema } from '../contracts/request.js';

// --- Signal detection constants (not exported) ---

/** D-02: package manager script prefixes that always qualify as codebase intent */
const PKG_MANAGER_RE = /^(npm|pnpm|yarn)\s+(test|build|lint|run|start|dev|check|format|typecheck)\b/i;

/** D-01: file path signals — path separator, file extension token, or known source dirs */
const FILE_PATH_RE = /(?:\/|\.(?:ts|tsx|js|jsx|py|go|rs|rb|java|sh|md|json|yaml|yml|toml|env)\b|(?:src|lib|tests?|dist|build)\/)/;

/** Filesystem operation keywords */
const FILESYSTEM_KEYWORDS_RE = /\b(?:rename|move|find file|copy|delete|rm|mv|cp|ls|list files?|image|folder|directory|\.(?:png|jpg|jpeg|gif|svg|pdf|docx?|xlsx?|csv|zip|tar|gz))\b/i;

/** Shell command signals: pipes, flags, common command prefixes */
const SHELL_COMMAND_RE = /(?:\||\bls\b|\becho\b|\bgrep\b|\bawk\b|\bsed\b|\bcurl\b|\bwget\b|\bssh\b|\bcat\b|\bhead\b|\btail\b|\bcd\b|\bpwd\b|\bmkdir\b|\btouch\b|\bchmod\b|-{1,2}[a-zA-Z])/;

export function classifyIntent(request: NormalizedRequest): IntentDecision {
  const query = request.lbuffer.trim();

  if (query.length === 0) {
    return { intent: 'unknown', confidence: 1, signals: ['empty-query'] };
  }

  // D-03: git prefix → shell-command + git-prefix signal (for provider gating in pipeline)
  if (/^git\s/.test(query)) {
    return { intent: 'shell-command', confidence: 1, signals: ['git-prefix'] };
  }

  // D-02: package manager scripts → codebase even without file path
  if (PKG_MANAGER_RE.test(query)) {
    return { intent: 'codebase', confidence: 1, signals: ['pkg-manager-script'] };
  }

  // D-01: file path signal required for codebase classification
  if (FILE_PATH_RE.test(query)) {
    return { intent: 'codebase', confidence: 0.9, signals: ['file-path-signal'] };
  }

  // Filesystem operations (before generic shell-command check)
  if (FILESYSTEM_KEYWORDS_RE.test(query)) {
    return { intent: 'filesystem', confidence: 0.85, signals: ['filesystem-keyword'] };
  }

  // Shell command signals
  if (SHELL_COMMAND_RE.test(query)) {
    return { intent: 'shell-command', confidence: 0.8, signals: ['shell-command-signal'] };
  }

  // Default fallback
  return { intent: 'general', confidence: 0.5, signals: ['no-strong-signal'] };
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| shell → Node process | `lbuffer` / `rbuffer` strings arrive from the zsh widget via the request JSON file — untrusted user input |
| contract parsing | `normalizedRequestSchema.parse(...)` runs on data from the file system |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | `src/contracts/request.ts` — `lbuffer` field | mitigate | Zod schema validation coerces and rejects malformed payloads; no eval or shell exec on query text in this plan |
| T-02-02 | Information Disclosure | `src/intent/router.ts` — signal array | accept | Signals are debug metadata emitted only to the debug log file, not returned to the shell or any external service |
| T-02-03 | Denial of Service | regex matching on untrusted `lbuffer` | mitigate | All regexes are linear-complexity anchored patterns with no catastrophic backtracking; `lbuffer` is a single-line ZLE buffer bounded by TTY width |
</threat_model>
