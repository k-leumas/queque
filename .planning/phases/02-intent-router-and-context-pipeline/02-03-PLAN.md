  <behavior>
    All four registries share identical structural behavior:
    - `register(descriptor)` — adds to internal Map; throws if id already exists, error message format: `<EntityType> already registered: "<id>"`
    - `get(id)` — returns descriptor or `undefined`
    - `list()` — returns `Array.from(registry.values())`
    - `clear()` — empties the Map; exported for test isolation ONLY; marked with `/** @internal */` JSDoc tag and a comment `// For test isolation only — do not call in production code`

    With `clear()` available, tests do NOT need `vi.resetModules()`. Use `beforeEach(() => { clearContextProviders(); })` instead — cleaner and avoids module-graph churn.

    Descriptor types for each registry:
    - context-providers: `ContextProvider` (from `src/context/provider.ts`)
    - provider-backends: `ProviderBackendDescriptor` — `{ id: string; name: string; description: string }` (STUB shape for Phase 2)
    - shell-adapters: `ShellAdapterDescriptor` — `{ id: string; shell: string; description: string }` (STUB shape; zsh is the Phase 2 built-in)
    - storage-hooks: `StorageHookDescriptor` — `{ id: string; description: string }` (STUB shape; noop and memory are Phase 2 built-ins)

    3. Gives a one-line example of the future use case

import { type ContextProvider } from '../context/provider.js';

const registry = new Map<string, ContextProvider>();

export function registerContextProvider(descriptor: ContextProvider): void {
  if (registry.has(descriptor.id)) {
    throw new Error(`Context provider already registered: "${descriptor.id}"`);
  }
  registry.set(descriptor.id, descriptor);
}

export function getContextProvider(id: string): ContextProvider | undefined {
  return registry.get(id);
}

export function listContextProviders(): ContextProvider[] {
  return Array.from(registry.values());
}

/**
 * @internal
 * Empties the registry. For test isolation only — do not call in production code.
 */
export function clearContextProviders(): void {
  registry.clear();
}

import { registerContextProvider, listContextProviders } from './context-providers.js';
import { registerShellAdapter } from './shell-adapters.js';
import { registerStorageHook } from './storage-hooks.js';
import { gitContextProvider } from '../context/providers/git-context.js';
import { filesystemContextProvider } from '../context/providers/filesystem-context.js';

let bootstrapped = false;

/**
 * Registers all Phase 2 built-in descriptors into their respective registries.
 *
 * Call once at process startup (called by gatherContext on first use).
 * Idempotent — safe to call multiple times; subsequent calls are no-ops.
 *
 * This is the single source of truth for what built-ins exist in Phase 2.
 * To add a new built-in, add it here — do not use module-load side effects.
 *
 * To inspect what is registered at runtime:
 *   import { listContextProviders } from './context-providers.js';
 *   console.log(listContextProviders().map(p => p.id));
 */
export function bootstrapBuiltins(): void {
  if (bootstrapped) return;
  bootstrapped = true;

  // Context providers — exercised by gatherContext on every request
  registerContextProvider(gitContextProvider);
  registerContextProvider(filesystemContextProvider);

  // Shell adapters — Phase 2 stub; zsh is the only built-in adapter
  registerShellAdapter({ id: 'zsh', shell: 'zsh', description: 'zsh ZLE widget adapter (Phase 2 built-in)' });

  // Storage hooks — Phase 2 stubs; no runtime behavior uses these yet
  registerStorageHook({ id: 'noop', description: 'No-op storage hook (default — no persistence)' });
  registerStorageHook({ id: 'memory', description: 'In-memory storage hook (test/ephemeral use)' });
}

/**
 * @internal
 * Resets the bootstrapped flag. For test isolation only — do not call in production code.
 */
export function resetBootstrap(): void {
  bootstrapped = false;
}

```typescript
export async function gatherContext(request: NormalizedRequest): Promise<ContextEnvelope> {
  // Ensure built-ins are registered (idempotent — no-op after first call)
  bootstrapBuiltins();

  void appendDebugLog('context', 'pipeline start', { lbuffer: request.lbuffer, intent: request.intent });

  // Pass 1: base context — always present
  const base = buildBaseContext(request);

  // Re-classify from the query to get fresh signals
  const decision = classifyIntent(request);
  void appendDebugLog('context', 'intent decision', { intent: decision.intent, signals: decision.signals });

  // Pass 2: intent-gated extras — reads from registry
  const providers = listContextProviders();
  const extras = [];
  for (const provider of providers) {
    if (!provider.intents.includes(decision.intent) && !provider.intents.includes('*')) {
      continue;
    }
    try {
      const chunk = await provider.gather({ base, decision });
      if (chunk !== null) {
        extras.push(chunk);
        void appendDebugLog('context', 'provider chunk gathered', { providerId: provider.id, kind: chunk.kind });
      }
    } catch (error) {
      void appendDebugLog('context', 'provider error', {
        providerId: provider.id,
        message: error instanceof Error ? error.message : String(error),
      });
    }
  }

  const envelope: ContextEnvelope = { base, extras };
  void appendDebugLog('context', 'envelope assembled', { extraCount: extras.length });
  return envelope;
}
```
