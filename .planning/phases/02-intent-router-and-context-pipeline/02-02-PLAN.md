**src/context/provider.ts** — ContextProvider interface and GatherContextInput:

```typescript
import { type BaseContext, type IntentDecision, type ContextChunk } from '../contracts/request.js';

/**
 * Input passed to every context provider's gather function.
 */
export interface GatherContextInput {
  base: BaseContext;
  decision: IntentDecision;
}

/**
 * A context provider gathers extra context relevant to a specific set of intents.
 * Built-in providers register through the context-providers registry (Plan 02-03).
 * The intents field declares which intent categories trigger this provider.
 * Use ['*'] to run the provider unconditionally (base context uses this internally).
 */
export interface ContextProvider {
  id: string;
  intents: string[];
  gather(input: GatherContextInput): Promise<ContextChunk | null>;
}
```

Note: ContextProvider is a TypeScript `interface` (not a Zod schema) because function types cannot be validated at runtime — per PATTERNS.md "No Analog Found" section.

**src/context/providers/git-context.ts** — wraps detectVcsContext + robust changedFiles parsing:

```typescript
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { type ContextProvider, type GatherContextInput } from '../provider.js';
import { type ContextChunk } from '../../contracts/request.js';
import { detectVcsContext } from '../../shared/vcs-context.js';
import { appendDebugLog } from '../../shared/debug-log.js';

const execFileAsync = promisify(execFile);

/** Subprocess timeout for git operations to prevent hang on slow/network-mounted filesystems. */
const GIT_TIMEOUT_MS = 5000;

/**
 * Parses one line of `git status --porcelain` output into a file path.
 *
 * Porcelain format: XY PATH or XY ORIG -> PATH (for renames/copies).
 * X = index status char, Y = working-tree status char.
 * Special cases:
 *   - Rename/Copy (R/C): "R  old-name -> new-name" — we want the destination (after ' -> ')
 *   - Untracked:         "?? path"
 *   - Merge conflict:    "UU path", "AA path", "DD path" — same as normal XY
 *   - Quoted paths:      XY "path with spaces" — strip surrounding double-quotes
 *
 * Returns the destination path only (never the rename source).
 * Returns null if the line cannot be parsed.
 */
function parsePorcelainLine(line: string): string | null {
  if (line.length < 4) return null;

  const rest = line.slice(3); // strip the two status chars and the space separator

  // Rename / copy: contains ' -> ' separator — take the destination
  const renameSep = ' -> ';
  const renameIdx = rest.indexOf(renameSep);
  if (renameIdx !== -1) {
    const dest = rest.slice(renameIdx + renameSep.length);
    return stripQuotes(dest);
  }

  return stripQuotes(rest);
}

function stripQuotes(s: string): string {
  if (s.startsWith('"') && s.endsWith('"')) {
    return s.slice(1, -1);
  }
  return s;
}

async function getChangedFiles(cwd: string): Promise<string[]> {
  try {
    const { stdout } = await execFileAsync('git', ['-C', cwd, 'status', '--porcelain'], {
      encoding: 'utf8',
      timeout: GIT_TIMEOUT_MS,
    });
    return stdout
      .split('\n')
      .filter(Boolean)
      .map(parsePorcelainLine)
      .filter((f): f is string => f !== null);
  } catch {
    return [];
  }
}

/**
 * Git context provider — runs for 'codebase' intent and for 'shell-command'
 * queries that start with 'git ' (D-09). The pipeline handles the git-prefix
 * gating via the IntentDecision signals; this provider declares both intents.
 *
 * Gathers: branch, dirty flag, and changed file names (no content — D-05/D-06).
 */
export const gitContextProvider: ContextProvider = {
  id: 'git-context',
  intents: ['codebase', 'shell-command'],
  async gather(input: GatherContextInput): Promise<ContextChunk | null> {
    // D-09: for shell-command, only run when the git-prefix signal is present
    if (
      input.decision.intent === 'shell-command' &&
      !input.decision.signals.includes('git-prefix')
    ) {
      return null;
    }

    try {
      const vcs = await detectVcsContext(input.base.cwd);
      if (vcs.kind === 'none') {
        return null;
      }
      const changedFiles = await getChangedFiles(input.base.cwd);
      void appendDebugLog('context', 'git provider gathered', {
        branch: vcs.branch,
        dirty: vcs.dirty,
        changedFiles: changedFiles.length,
      });
      return {
        kind: 'git',
        payload: {
          cwd: vcs.cwd,
          root: vcs.root,
          branch: vcs.branch,
          dirty: vcs.dirty,
          changedFiles,
        },
      };
    } catch (error) {
      void appendDebugLog('context', 'git provider failed', {
        message: error instanceof Error ? error.message : String(error),
      });
      return null;
    }
  },
};
```

**src/context/providers/filesystem-context.ts** — string parse only, no I/O:

```typescript
import { type ContextProvider, type GatherContextInput } from '../provider.js';
import { type ContextChunk } from '../../contracts/request.js';

/** Matches token.ext patterns in query text — e.g. hero-banner.png, report.pdf */
const FILENAME_RE = /\b[\w.-]+\.\w{2,6}\b/g;

/**
 * Filesystem context provider — runs for 'filesystem' intent only (D-11).
 * String parse only — no disk I/O, no directory listing (D-10).
 * Extracts an apparent filename from the query if one is present.
 */
export const filesystemContextProvider: ContextProvider = {
  id: 'filesystem-context',
  intents: ['filesystem'],
  async gather(input: GatherContextInput): Promise<ContextChunk | null> {
    const match = input.base.queryText.match(FILENAME_RE);
    return {
      kind: 'filesystem',
      payload: {
        cwd: input.base.cwd,
        apparentFilename: match?.[0] ?? null,
      },
    };
  },
};
```

```typescript
import { type NormalizedRequest, type ContextEnvelope } from '../contracts/request.js';
import { buildBaseContext } from './base-context.js';
import { classifyIntent } from '../intent/router.js';
import { gitContextProvider } from './providers/git-context.js';
import { filesystemContextProvider } from './providers/filesystem-context.js';
import { appendDebugLog } from '../shared/debug-log.js';

/**
 * Hard-coded built-in provider list for Phase 2.
 * Plan 02-03 moves this into the context-providers registry.
 */
const BUILTIN_PROVIDERS = [gitContextProvider, filesystemContextProvider];

/**
 * Two-pass context pipeline:
 *   Pass 1 — base context (always, synchronous, no I/O)
 *   Pass 2 — intent-gated extras (only providers whose intents match)
 *
 * Providers returning null are silently skipped.
 * Provider errors return null (logged, not thrown).
 */
export async function gatherContext(request: NormalizedRequest): Promise<ContextEnvelope> {
  void appendDebugLog('context', 'pipeline start', { lbuffer: request.lbuffer, intent: request.intent });

  // Pass 1: base context — always present
  const base = buildBaseContext(request);

  // Re-classify from the query to get fresh signals (pipeline owns classification)
  const decision = classifyIntent(request);
  void appendDebugLog('context', 'intent decision', { intent: decision.intent, signals: decision.signals });

  // Pass 2: intent-gated extras
  const extras = [];
  for (const provider of BUILTIN_PROVIDERS) {
    if (!provider.intents.includes(decision.intent) && !provider.intents.includes('*')) {
      continue;
    }
    try {
      const chunk = await provider.gather({ base, decision });
      if (chunk !== null) {
        extras.push(chunk);
        void appendDebugLog('context', 'provider chunk gathered', { providerId: provider.id, kind: chunk.kind });
      }
    } catch (error) {
      void appendDebugLog('context', 'provider error', {
        providerId: provider.id,
        message: error instanceof Error ? error.message : String(error),
      });
    }
  }

  const envelope: ContextEnvelope = { base, extras };
  void appendDebugLog('context', 'envelope assembled', { extraCount: extras.length });
  return envelope;
}
```
